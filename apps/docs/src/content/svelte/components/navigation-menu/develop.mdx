## Import

```ts
import { NavigationMenu } from "@pisagor/svelte";
```

## Anatomy

```tsx
<NavigationMenu>
  <NavigationMenu.List>
    <NavigationMenu.Item>
      <NavigationMenu.Link />
    </NavigationMenu.Item>
  </NavigationMenu.List>
</NavigationMenu>
```

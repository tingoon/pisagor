## Import

```ts
import { TreeView } from "@pisagor/svelte";
```

## Anatomy

```tsx
<TreeView collection={collection}>
  <TreeView.Tree>
    <TreeView.NodeProvider>
      <TreeView.Branch>
        <TreeView.BranchControl />
        <TreeView.BranchContent />
      </TreeView.Branch>
      <TreeView.Item>
        <TreeView.ItemText />
      </TreeView.Item>
    </TreeView.NodeProvider>
  </TreeView.Tree>
</TreeView>
```

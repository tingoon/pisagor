## Import

```ts
import { ColorPicker } from "@pisagor/svelte";
```

## Anatomy

```tsx
<ColorPicker>
  <ColorPicker.Control>
    <ColorPicker.Trigger>
      <ColorPicker.SwatchPreview />
    </ColorPicker.Trigger>
    <ColorPicker.Input />
  </ColorPicker.Control>
  <ColorPicker.Content>
    <ColorPicker.Area>
      <ColorPicker.AreaThumb />
    </ColorPicker.Area>
    <ColorPicker.View>
      <ColorPicker.EyeDropperTrigger />
      <ColorPicker.ChannelSlider />
    </ColorPicker.View>
  </ColorPicker.Content>
</ColorPicker>
```

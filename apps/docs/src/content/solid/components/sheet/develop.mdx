## Import

```tsx
import { Sheet } from "@pisagor/solid";
```

## Anatomy

```tsx
<Sheet>
  <Sheet.Trigger />
  <Sheet.Content>
    <Sheet.Header>
      <Sheet.Title />
      <Sheet.Description />
    </Sheet.Header>
    <Sheet.Body />
    <Sheet.Footer>
      <Sheet.CloseTrigger />
    </Sheet.Footer>
  </Sheet.Content>
</Sheet>
```

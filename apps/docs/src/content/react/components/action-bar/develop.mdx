## Import

```tsx
import { ActionBar } from "@pisagor/react";
```

## Anatomy

```tsx
<ActionBar>
  <ActionBar.Trigger />
  <ActionBar.Content>
    <ActionBar.Value />
    <ActionBar.Separator />
    <ActionBar.Body />
    <ActionBar.Close />
  </ActionBar.Content>
</ActionBar>
```

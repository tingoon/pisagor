## Import

```tsx
import { Drawer } from "@pisagor/react";
```

## Anatomy

```tsx
<Drawer>
  <Drawer.Trigger />
  <Drawer.Content>
    <Drawer.ContentInner>
      <Drawer.Header />
      <Drawer.Body />
    </Drawer.ContentInner>
    <Drawer.Footer>
      <Drawer.CloseTrigger />
    </Drawer.Footer>
  </Drawer.Content>
</Drawer>
```

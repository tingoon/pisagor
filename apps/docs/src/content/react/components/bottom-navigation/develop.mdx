## Import

```tsx
import { BottomNavigation } from "@pisagor/react";
```

## Anatomy

```tsx
<BottomNavigation>
  <BottomNavigation.List>
    <BottomNavigation.Item>
      <BottomNavigation.ItemIcon />
      <BottomNavigation.ItemLabel />
    </BottomNavigation.Item>
  </BottomNavigation.List>
</BottomNavigation>
```

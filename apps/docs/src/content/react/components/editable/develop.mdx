## Import

```tsx
import { Editable } from "@pisagor/react";
```

## Anatomy

```tsx
<Editable>
  <Editable.Area>
    <Editable.Input />
    <Editable.Preview />
  </Editable.Area>
  <Editable.Control>
    <Editable.CancelTrigger />
    <Editable.SubmitTrigger />
  </Editable.Control>
</Editable>
```

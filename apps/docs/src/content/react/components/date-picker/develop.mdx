## Import

```tsx
import { DatePicker } from "@pisagor/react";
```

## Anatomy

```tsx
<DatePicker>
  <DatePicker.Trigger>
    <DatePicker.ValueText />
  </DatePicker.Trigger>
  <DatePicker.Content />
</DatePicker>
```

## Import

```tsx
import { TagsInput } from "@pisagor/react";
```

## Anatomy

```tsx
<TagsInput>
  <TagsInput.Context>
    {({ value }) => value.map((tag, index) => <TagsInput.Item />)}
  </TagsInput.Context>
</TagsInput>
```

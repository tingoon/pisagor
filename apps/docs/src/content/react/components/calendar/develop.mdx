## Import

```tsx
import { Calendar } from "@pisagor/react";
```

## Anatomy

```tsx
<Calendar>
  <Calendar.ViewControl>
    <Calendar.PrevTrigger />
    <Calendar.ViewDate />
    <Calendar.NextTrigger />
  </Calendar.ViewControl>
  <Calendar.Table>
    <Calendar.WeekDays />
    <Calendar.TableDays />
  </Calendar.Table>
</Calendar>
```

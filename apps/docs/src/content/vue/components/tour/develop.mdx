## Import

```ts
import { Tour } from "@pisagor/vue";
```

## Anatomy

```vue
<Tour>
  <Tour.Trigger />
  <Tour.Content>
    <Tour.Header>
      <Tour.ProgressText />
      <Tour.Title />
      <Tour.Description />
    </Tour.Header>
    <Tour.Footer>
      <Tour.PreviousStep />
      <Tour.NextStep />
    </Tour.Footer>
  </Tour.Content>
</Tour>
```

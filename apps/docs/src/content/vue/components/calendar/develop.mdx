## Import

```ts
import { Calendar } from "@pisagor/vue";
```

## Anatomy

```vue
<Calendar>
  <Calendar.ViewControl>
    <Calendar.PrevTrigger />
    <Calendar.ViewDate />
    <Calendar.NextTrigger />
  </Calendar.ViewControl>
  <Calendar.Table>
    <Calendar.WeekDays />
    <Calendar.TableDays />
  </Calendar.Table>
</Calendar>
```

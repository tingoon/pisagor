## Import

```ts
import { Drawer } from "@pisagor/vue";
```

## Anatomy

```vue
<Drawer>
  <Drawer.Trigger />
  <Drawer.Content>
    <Drawer.ContentInner>
      <Drawer.Header />
      <Drawer.Body />
    </Drawer.ContentInner>
    <Drawer.Footer>
      <Drawer.CloseTrigger />
    </Drawer.Footer>
  </Drawer.Content>
</Drawer>
```

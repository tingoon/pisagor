## Import

```ts
import { Editable } from "@pisagor/vue";
```

## Anatomy

```vue
<Editable>
  <Editable.Area>
    <Editable.Input />
    <Editable.Preview />
  </Editable.Area>
  <Editable.Control>
    <Editable.CancelTrigger />
    <Editable.SubmitTrigger />
  </Editable.Control>
</Editable>
```
